import { chunk } from 'lodash';
import { scheduleOnUI } from 'react-native-worklets';

scheduleOnUI(() => {
  'worklet';
  const groups = chunk([1, 2, 3, 4], 2);
  console.log(groups); // [[1, 2], [3, 4]]

  // No changes to Lodash needed 🤯
});
